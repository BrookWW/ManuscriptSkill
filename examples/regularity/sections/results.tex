\section{An estimate}
\begin{theorem}\label{thm:regularity}
Every continuous function on $[0,1]$ is Lipschitz continuous.
\end{theorem}
\begin{proof}
Continuity on a compact interval gives uniform continuity. Therefore the
ratio $|\f(x)-\f(y)|/|x-y|$ is bounded, as required.
\end{proof}
\begin{theorem}\label{thm:linear}
For $\f(x)=2x+1$, $|\f(x)-\f(y)|=2|x-y|$.
\end{theorem}
\begin{proof}
Subtract the two values and take absolute values.
\end{proof}
